# Setup Guide — AI Integration (OpenAI + Gemini via Port Abstraction)

> Full setup for Eduvia's AI layer using the Port/Adapter pattern so providers are swappable without touching domain code.

---

## Architecture Decision

AI providers are abstracted behind an `IAIPort` interface. The active provider is selected per request (or per job) — not globally fixed. This allows:
- Using OpenAI GPT-4o for report summaries and Gemini for question generation
- A/B testing providers without code changes
- Graceful fallback: if OpenAI fails, retry with Gemini

All AI calls are async-only — triggered via BullMQ `ai-events` queue. AI is NEVER called synchronously in a request handler.

## 2. AI Port Interface

```typescript
// src/modules/ai/domain/ports/ai.port.ts

export interface AIGenerateParams {
  systemPrompt: string;
  userPrompt: string;
  maxTokens?: number;
  temperature?: number;   // 0.0 = deterministic, 1.0 = creative
  jsonMode?: boolean;     // request structured JSON output
}

export interface AIGenerateResult {
  content: string;
  tokensUsed: number;
  model: string;
  provider: 'openai' | 'gemini';
}

abstract class IAIPort {
  abstract generate(params: AIGenerateParams): Promise<AIGenerateResult>;
  abstract readonly provider: 'openai' | 'gemini';
  abstract readonly model: string;
}

export { IAIPort };
```

## 3. OpenAI Adapter

```typescript
// src/modules/ai/infrastructure/adapters/openai.adapter.ts

import { Injectable, Logger } from '@nestjs/common';
import OpenAI from 'openai';
import { IAIPort, AIGenerateParams, AIGenerateResult } from '../../domain/ports/ai.port';

@Injectable()
export class OpenAIAdapter implements IAIPort {
  readonly provider = 'openai' as const;
  readonly model = 'gpt-4o';
  private readonly logger = new Logger(OpenAIAdapter.name);

  private readonly client = new OpenAI({
    apiKey: process.env.OPENAI_API_KEY,
    maxRetries: 3,
    timeout: 30_000,
  });

  async generate(params: AIGenerateParams): Promise<AIGenerateResult> {
    const response = await this.client.chat.completions.create({
      model: this.model,
      max_tokens: params.maxTokens ?? 1000,
      temperature: params.temperature ?? 0.3,
      response_format: params.jsonMode ? { type: 'json_object' } : { type: 'text' },
      messages: [
        { role: 'system', content: params.systemPrompt },
        { role: 'user', content: params.userPrompt },
      ],
    });

    const content = response.choices[0]?.message?.content ?? '';
    const tokensUsed = response.usage?.total_tokens ?? 0;

    this.logger.debug(`OpenAI (${this.model}): ${tokensUsed} tokens used`);

    return { content, tokensUsed, model: this.model, provider: this.provider };
  }
}
```

## 4. Gemini Adapter

```typescript
// src/modules/ai/infrastructure/adapters/gemini.adapter.ts

import { Injectable, Logger } from '@nestjs/common';
import { GoogleGenerativeAI, HarmCategory, HarmBlockThreshold } from '@google/generative-ai';
import { IAIPort, AIGenerateParams, AIGenerateResult } from '../../domain/ports/ai.port';

@Injectable()
export class GeminiAdapter implements IAIPort {
  readonly provider = 'gemini' as const;
  readonly model = 'gemini-2.0-flash';
  private readonly logger = new Logger(GeminiAdapter.name);

  private readonly genAI = new GoogleGenerativeAI(process.env.GOOGLE_AI_API_KEY!);

  async generate(params: AIGenerateParams): Promise<AIGenerateResult> {
    const model = this.genAI.getGenerativeModel({
      model: this.model,
      safetySettings: [
        { category: HarmCategory.HARM_CATEGORY_HARASSMENT, threshold: HarmBlockThreshold.BLOCK_NONE },
        { category: HarmCategory.HARM_CATEGORY_HATE_SPEECH, threshold: HarmBlockThreshold.BLOCK_NONE },
      ],
      generationConfig: {
        maxOutputTokens: params.maxTokens ?? 1000,
        temperature: params.temperature ?? 0.3,
        responseMimeType: params.jsonMode ? 'application/json' : 'text/plain',
      },
      systemInstruction: params.systemPrompt,
    });

    const result = await model.generateContent(params.userPrompt);
    const content = result.response.text();

    // Gemini does not expose token count in a simple way — use metadata
    const tokensUsed = result.response.usageMetadata?.totalTokenCount ?? 0;

    this.logger.debug(`Gemini (${this.model}): ${tokensUsed} tokens used`);

    return { content, tokensUsed, model: this.model, provider: this.provider };
  }
}
```

## 5. AI Provider Factory

```typescript
// src/modules/ai/infrastructure/ai-provider.factory.ts

import { Injectable } from '@nestjs/common';
import { IAIPort } from '../domain/ports/ai.port';
import { OpenAIAdapter } from './adapters/openai.adapter';
import { GeminiAdapter } from './adapters/gemini.adapter';

@Injectable()
export class AIProviderFactory {
  constructor(
    private readonly openai: OpenAIAdapter,
    private readonly gemini: GeminiAdapter,
  ) {}

  get(provider: 'openai' | 'gemini'): IAIPort {
    return provider === 'openai' ? this.openai : this.gemini;
  }

  // Returns the default provider, with auto-fallback
  getDefault(): IAIPort {
    const defaultProvider = (process.env.AI_DEFAULT_PROVIDER ?? 'openai') as 'openai' | 'gemini';
    return this.get(defaultProvider);
  }
}
```

## 6. Prompt Template Service

```typescript
// src/modules/ai/application/services/prompt-template.service.ts

import { Injectable } from '@nestjs/common';
import { IAIPromptTemplateRepository } from '../../domain/repository/ai-prompt-template.repository';

@Injectable()
export class PromptTemplateService {
  constructor(
    private readonly templateRepository: IAIPromptTemplateRepository,
  ) {}

  async render(type: string, variables: Record<string, string>): Promise<{
    systemPrompt: string;
    userPrompt: string;
    version: string;
  }> {
    const template = await this.templateRepository.findActive(type);
    if (!template) throw new Error(`No active prompt template for type: ${type}`);

    // Replace {{variableName}} placeholders
    const userPrompt = template.userPromptTemplate.replace(
      /\{\{(\w+)\}\}/g,
      (_, key) => variables[key] ?? `[MISSING: ${key}]`,
    );

    return {
      systemPrompt: template.systemPrompt,
      userPrompt,
      version: template.version,
    };
  }
}
```

## 7. AI Workers (BullMQ Processors)

### Report Summary Worker

```typescript
// src/modules/ai/application/workers/report-summary.worker.ts

import { Processor, WorkerHost } from '@nestjs/bullmq';
import { Job } from 'bullmq';
import { Logger } from '@nestjs/common';
import { AIProviderFactory } from '../../infrastructure/ai-provider.factory';
import { PromptTemplateService } from '../services/prompt-template.service';
import { IAIJobRepository } from '../../domain/repository/ai-job.repository';
import { ILessonReportRepository } from '@modules/progress';
import { CommandBus } from '@nestjs/cqrs';
import { AttachAiReportDraftCommand } from '../commands';
import { AIJob } from '../../domain/model/AIJob';
import { AIJobId } from '../../domain/value-objects/ai-job-id.vo';

@Processor('ai-events')
export class AIEventWorker extends WorkerHost {
  private readonly logger = new Logger(AIEventWorker.name);

  constructor(
    private readonly providerFactory: AIProviderFactory,
    private readonly templateService: PromptTemplateService,
    private readonly aiJobRepository: IAIJobRepository,
    private readonly commandBus: CommandBus,
  ) {
    super();
  }

  async process(job: Job): Promise<void> {
    switch (job.name) {
      case 'LessonCompletedEvent':
        await this.handleReportSummary(job);
        break;
      case 'MasteryUpdatedEvent':
        await this.handleNextLessonSuggestion(job);
        break;
      case 'QuestionGenerationRequestedEvent':
        await this.handleQuestionGeneration(job);
        break;
    }
  }

  private async handleReportSummary(job: Job): Promise<void> {
    const { lessonId, orgId } = job.data.payload;

    // Create an AIJob record for auditability
    const aiJob = AIJob.create({
      id: AIJobId.create(),
      type: 'report_summary',
      relatedEntityId: lessonId,
      relatedEntityType: 'lesson',
      orgId,
      provider: 'openai',
      model: 'gpt-4o',
      promptVersion: '',       // filled after template lookup
      inputPayload: job.data.payload,
      correlationId: job.data.correlationId,
    });

    await this.aiJobRepository.save(aiJob);

    try {
      // Build the prompt with lesson data
      const { systemPrompt, userPrompt, version } = await this.templateService.render(
        'report_summary',
        {
          studentName: job.data.payload.studentName,
          teacherName: job.data.payload.teacherName,
          subject: job.data.payload.subject,
          topicsCovered: job.data.payload.topicsCovered?.join(', ') ?? '',
          boardContent: JSON.stringify(job.data.payload.boardSections ?? []),
          assessmentSummary: JSON.stringify(job.data.payload.assessmentSummary ?? {}),
        },
      );

      aiJob.markProcessing(version);
      await this.aiJobRepository.save(aiJob);

      const ai = this.providerFactory.getDefault();
      const result = await ai.generate({
        systemPrompt,
        userPrompt,
        maxTokens: 800,
        temperature: 0.4,
        jsonMode: true,
      });

      const parsedOutput = JSON.parse(result.content) as {
        summary: string;
        strengths: string[];
        areasForImprovement: string[];
        nextSteps: string;
      };

      aiJob.complete(parsedOutput, result.tokensUsed);
      await this.aiJobRepository.save(aiJob);

      // Attach the AI draft to the lesson report
      await this.commandBus.execute(
        new AttachAiReportDraftCommand({
          lessonId,
          orgId,
          aiSummary: parsedOutput.summary,
          strengths: parsedOutput.strengths,
          areasForImprovement: parsedOutput.areasForImprovement,
          nextSteps: parsedOutput.nextSteps,
          correlationId: job.data.correlationId,
        }),
      );

    } catch (error) {
      this.logger.error(`AI report summary failed for lesson ${lessonId}`, error);
      aiJob.fail((error as Error).message);
      await this.aiJobRepository.save(aiJob);
      // Job failure emits AIJobFailedEvent → notification to teacher to write manually
      throw error; // Let BullMQ retry
    }
  }

  private async handleQuestionGeneration(job: Job): Promise<void> {
    const { curriculumNodeId, questionType, count, orgId } = job.data.payload;

    const { systemPrompt, userPrompt, version } = await this.templateService.render(
      'question_generation',
      {
        nodeName: job.data.payload.nodeName,
        nodeDescription: job.data.payload.nodeDescription ?? '',
        gradeLevel: job.data.payload.gradeLevel ?? '',
        curriculumName: job.data.payload.curriculumName,
        questionType,
        difficultyLevel: String(job.data.payload.difficultyLevel ?? 3),
        count: String(count),
        existingQuestions: job.data.payload.existingQuestions?.join('\n') ?? '',
      },
    );

    const ai = this.providerFactory.getDefault();
    const result = await ai.generate({
      systemPrompt,
      userPrompt,
      maxTokens: 2000,
      temperature: 0.6,
      jsonMode: true,
    });

    const { questions } = JSON.parse(result.content) as { questions: unknown[] };

    // Create each generated question in the DB via CommandBus
    for (const q of questions) {
      await this.commandBus.execute(/* CreateQuestionFromAICommand(q, orgId, curriculumNodeId, 'openai', ...) */);
    }
  }

  private async handleNextLessonSuggestion(_job: Job): Promise<void> {
    // Similar pattern: build prompt from mastery graph, call AI, store suggestion
  }
}
```

```typescript
// AI workers need longer timeouts and lower concurrency
@Processor('ai-events', {
  concurrency: 2,              // max 2 AI jobs at once (rate limit protection)
  limiter: {
    max: 10,                   // max 10 jobs per...
    duration: 60_000,          // ...per 60 seconds (protect against OpenAI rate limits)
  },
})
```
